cmake_minimum_required(VERSION 3.21)
project(PacketFlow VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(PACKETFLOW_GUI "Build the native Qt Quick application" ON)
find_package(Threads REQUIRED)
find_path(PCAP_INCLUDE_DIR pcap/pcap.h REQUIRED)
find_library(PCAP_LIBRARY pcap REQUIRED)
add_library(packetcore src/packet.cpp src/demo.cpp src/engine.cpp src/capture.cpp)
target_include_directories(packetcore PUBLIC src PRIVATE ${PCAP_INCLUDE_DIR})
target_link_libraries(packetcore PUBLIC ${PCAP_LIBRARY} Threads::Threads)
include(CTest)
if(BUILD_TESTING)
  add_executable(packet_tests tests/core.cpp)
  target_link_libraries(packet_tests PRIVATE packetcore)
  add_test(NAME packet_core COMMAND packet_tests)
endif()
if(PACKETFLOW_GUI)
  find_package(Qt6 6.5 REQUIRED COMPONENTS Quick QuickControls2)
  qt_standard_project_setup()
  find_path(MAXMINDDB_INCLUDE_DIR maxminddb.h REQUIRED)
  find_library(MAXMINDDB_LIBRARY maxminddb REQUIRED)
  add_library(packetui STATIC src/session.cpp src/map.cpp src/geoip.cpp)
  target_include_directories(packetui PRIVATE ${MAXMINDDB_INCLUDE_DIR})
  target_link_libraries(packetui PRIVATE ${MAXMINDDB_LIBRARY})
  target_link_libraries(packetui PUBLIC packetcore Qt6::Quick Qt6::QuickControls2)
  set(packet_qml ui/Main.qml ui/ToolButtonPF.qml ui/Glyph.qml ui/Inspector.qml
      ui/Sheets.qml ui/Legend.qml ui/Timeline.qml ui/Packets.qml)
  qt_add_executable(PacketFlow MACOSX_BUNDLE src/main.cpp)
  set(PACKETFLOW_GEOIP_DATABASE "" CACHE FILEPATH "Optional City or Country MMDB to include in the macOS app")
  if(APPLE AND PACKETFLOW_GEOIP_DATABASE)
    get_filename_component(geoip_file "${PACKETFLOW_GEOIP_DATABASE}" ABSOLUTE)
    get_filename_component(geoip_name "${geoip_file}" NAME)
    target_sources(PacketFlow PRIVATE "${geoip_file}")
    set_source_files_properties("${geoip_file}" PROPERTIES MACOSX_PACKAGE_LOCATION "Resources/GeoIP")
    target_sources(PacketFlow PRIVATE docs/db-ip-attribution.txt)
    set_source_files_properties(docs/db-ip-attribution.txt PROPERTIES MACOSX_PACKAGE_LOCATION "Resources/GeoIP")
    target_compile_definitions(packetui PRIVATE PACKETFLOW_BUNDLED_MMDB="${geoip_name}")
  endif()
  qt_add_resources(PacketFlow interface PREFIX "/" FILES ${packet_qml})
  target_link_libraries(PacketFlow PRIVATE packetui)
  set_target_properties(PacketFlow PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER io.packetflow.desktop)
  if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(session_tests tests/session.cpp)
    qt_add_resources(session_tests test_interface PREFIX "/" FILES ${packet_qml})
    target_link_libraries(session_tests PRIVATE packetui Qt6::Test)
    add_test(NAME packet_session COMMAND session_tests)
    set_tests_properties(packet_session PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
  endif()
endif()
